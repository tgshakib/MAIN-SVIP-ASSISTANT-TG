import os
import secrets
from flask import Flask, send_file, abort, request

app = Flask(__name__)

# One-time token — printed to console on startup
# Anyone without this token gets a 403
DOWNLOAD_TOKEN = secrets.token_urlsafe(16)

@app.route("/")
def index():
    token = request.args.get("token", "")
    if token != DOWNLOAD_TOKEN:
        abort(403)
    zip_path = os.path.join(os.path.dirname(__file__), "mybot_backup.zip")
    if not os.path.exists(zip_path):
        abort(404)
    return send_file(
        zip_path,
        as_attachment=True,
        download_name="mybot_backup.zip",
        mimetype="application/zip"
    )

@app.route("/health")
def health():
    return {"status": "ok"}, 200

if __name__ == "__main__":
    print("\n" + "="*55)
    print("  ✅ Download server is LIVE")
    print(f"  🔗 URL: http://0.0.0.0:8080/?token={DOWNLOAD_TOKEN}")
    print("  📦 File: mybot_backup.zip")
    print("  🔒 Token required — copy the full URL above")
    print("="*55 + "\n")
    app.run(host="0.0.0.0", port=8080)
PYEOF

echo "✅ STEP 6 DONE — download_server.py written"
